"""Launch g4 once under the posted maintainer smoke-completion waiver."""
import datetime,hashlib,json,subprocess
from pathlib import Path
H=Path(__file__).resolve().parent;E=H/'evidence';J=H/'g4'
read=lambda p:json.loads(p.read_text())
freeze=read(H/'freeze-g4.json');post=read(H/'freeze-post.json')
assert post['freeze_sha256']==hashlib.sha256((H/'freeze-g4.json').read_bytes()).hexdigest()
assert not (J/'process.json').exists()
amendment=read(H/'g4-amendment.json');amendment_post=read(H/'amendment-post.json')
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
assert amendment_post['amendment_sha256']==sha(H/'g4-amendment.json')
assert amendment['freeze_sha256']==sha(H/'freeze-g4.json')
assert amendment['launcher_sha256']==sha(Path(__file__))
assert amendment['waived_check']=='smoke session completion only'
assert amendment['smoke_evidence']['provenance_hashes_verified'] is True
GH='/home/jason/Workspace/ViviDynamics/skills/skills/ship-issue/scripts/vgh'
for record,body_name,required in [(post,'freeze-on-51.md',post['freeze_sha256']),
                                 (amendment_post,'amendment-on-51.md',amendment_post['amendment_sha256'])]:
 comment_id=record['url'].rsplit('issuecomment-',1)[1]
 body=subprocess.check_output([GH,'api','repos/ViviDynamics/scrutare/issues/comments/'+comment_id,'--jq','.body'],text=True).rstrip('\n')
 assert hashlib.sha256((body+'\n').encode()).hexdigest()==record['body_sha256']
 assert required in body
 assert sha(H/body_name)==record['body_sha256']
for name,digest in amendment['frozen_files_sha256'].items():
 assert sha(H/name)==digest,name
for name,digest in freeze['files'].items():assert hashlib.sha256((J/name).read_bytes()).hexdigest()==digest,name
for name,digest in freeze['script_sha256'].items():assert hashlib.sha256((H/name).read_bytes()).hexdigest()==digest,name
subprocess.run(['/usr/bin/python3',str(H/'private-call.py'),'fallback-check.py',str(E/'fallback-check-prelaunch.json')],check=True)
end=datetime.datetime.now(datetime.timezone.utc);start=end-datetime.timedelta(hours=1)
fmt=lambda t:t.strftime('%Y-%m-%d %H:%M:%S')
subprocess.run(['/usr/bin/python3',str(H/'collect-window.py'),'prelaunch',fmt(start),fmt(end)],check=True)
subprocess.run(['systemd-run','--user','--unit=scrutare-glm-g4','--setenv=GLM_UNIT=scrutare-glm-g4.service',f'--property=WorkingDirectory={J}','/usr/bin/python3',str(H/'launch-detached.py'),str(J)],check=True)
for suffix,script in [('load','load-sampler.py'),('finish','finish-evidence.py')]:
 subprocess.run(['systemd-run','--user','--unit=scrutare-glm-g4-'+suffix,f'--property=WorkingDirectory={H}',f'--property=StandardOutput=append:{E/(suffix+"-service.log")}',f'--property=StandardError=append:{E/(suffix+"-service.log")}','/usr/bin/python3',str(H/script)],check=True)
(H/'launch-ledger.json').write_text(json.dumps({'launched_once':True,'schedule':'g4','freeze_url':post['url'],'observed_utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'no_idle_window':True,'evaluator_unit':'scrutare-glm-g4.service','load_unit':'scrutare-glm-g4-load.service','finish_unit':'scrutare-glm-g4-finish.service'},indent=2)+'\n')
