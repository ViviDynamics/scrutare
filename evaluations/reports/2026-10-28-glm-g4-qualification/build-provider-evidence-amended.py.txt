"""Publish only allowlisted spend fields, with stable redaction of personal aliases."""
import collections,datetime,hashlib,json,re,shlex,sys
from pathlib import Path
H=Path(__file__).resolve().parent;E=H/'evidence';out=Path(sys.argv[1])
read=lambda p:json.loads(p.read_text())
K=next(shlex.split(l.partition('=')[2].strip())[0] for l in Path('/home/user/Workspace/example/infrastructure/.env').read_text().splitlines() if l.startswith('LITELLM_MASTER_KEY='))
key_aliases={};other_agents={}
def safe_key(alias):
 if alias in ('proxy master key','client-a','client-b'):return alias
 return key_aliases.setdefault(alias,f'other key {len(key_aliases)+1}')
def safe_agent(agent):
 if agent in ('python-httpx2','python-httpx','opencode','claude-code','none'):return agent
 return other_agents.setdefault(agent,f'other client {len(other_agents)+1}')
def row_key(r):
 md=r.get('metadata') or {}
 if md.get('user_api_key')=='litellm_proxy_master_key' or r.get('api_key')=='litellm_proxy_master_key':return 'proxy master key'
 return md.get('user_api_key_alias') or 'unnamed other key'
def slim(r):
 md=r.get('metadata') or {}
 return {'start_utc':r.get('startTime'),'end_utc':r.get('endTime'),'agent':safe_agent((md.get('user_agent') or 'none').split('/')[0]),'key':safe_key(row_key(r)),'model_group':r['model_group'],'model':r['model'],'api_base':r.get('api_base') or '', 'status':r['status'],'prompt_tokens':r.get('prompt_tokens') or 0,'completion_tokens':r.get('completion_tokens') or 0,'spend':r.get('spend') or 0,'attempted_fallbacks':md.get('attempted_fallbacks'),'attempted_retries':md.get('attempted_retries')}
windows={}
for label in ('smoke','prelaunch','g4'):
 rows=E/f'{label}-rows.json';summary=read(E/f'{label}-spend-summary.json')
 windows[label]={'window_utc':summary['window_utc'],'read_utc':summary['read_utc'],'raw_rows_sha256_private':hashlib.sha256(rows.read_bytes()).hexdigest(),'rows':[slim(r) for r in read(rows)]}
samples=[]
for line in (E/'load-g4-samples.jsonl').read_text().splitlines():
 s=json.loads(line);by={}
 for k,v in (s.get('by_agent_key_group_base') or {}).items():
  agent,alias,group,base=k.split(' | ')
  by[f'{safe_agent(agent)} | {safe_key(alias)} | {group} | {base}']=v
 samples.append({**{k:v for k,v in s.items() if k!='by_agent_key_group_base'},'by_agent_key_group_base':by})
doc={'schema_version':1,'windows':windows,'samples_15min':samples,
 'load_limits':'Each live sample is a lower bound: requests still in flight appear only when logged. The full-window re-read is authoritative for rows persisted by its recorded read time; other clients still in flight can arrive later. Traffic bypassing LiteLLM is invisible. python-httpx2 is nare on the shared proxy master key; smoke window has no qualification requests, g4 window has no smoke requests. Agent families omit version suffixes, as in c1.',
 'alias_redaction':'Only proxy master key, client-a and client-b aliases are retained; all other aliases and nonstandard agents use stable numbered labels. No key/hash, email, IP client address, request content or personal alias is published.',
 'launch':{k:v for k,v in read(H/'g4/process.json').items() if k in ('started_utc','systemd_unit','source_commit','expected_runs','label','model_alias','holdout_used')},
 'ended':read(E/'run-completion.json'),
 'smoke_activation_verification':read(E/'smoke-activation-verification.json')}
for field,name in [('router_before_smoke','fallback-check-smoke.json'),('router_prelaunch','fallback-check-prelaunch.json'),('router_after','fallback-check-after.json')]:
 d=read(E/name);doc[field]={k:d[k] for k in ('read_utc','ok','model_group_alias','fallback_fields','wildcard_groups','glm_deployments','num_retries','timeout')}
text=json.dumps(doc,indent=1,sort_keys=True)+'\n'
assert K not in text and hashlib.sha256(K.encode()).hexdigest() not in text
assert not re.search(r'[\w.+-]+@[\w.-]+\.[a-z]{2,}',text)
assert '/home/' not in text and '/tmp/' not in text
(out/'provider-evidence.json').write_text(text)
print(json.dumps({'windows':{k:len(v['rows']) for k,v in windows.items()},'samples':len(samples),'redacted_alias_count':len(key_aliases)}))
