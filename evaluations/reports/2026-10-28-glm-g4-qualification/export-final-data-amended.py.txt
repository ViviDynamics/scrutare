"""Export the single retained g4 job after final evidence is ready. Never calls a model."""
import hashlib,json,shutil,subprocess,time
from pathlib import Path
H=Path(__file__).resolve().parent;J=H/'g4'
W=Path('/home/jason/Workspace/ViviDynamics/scrutare-99-glm-g4')
P=W/'evaluations/reports/2026-10-28-glm-g4-qualification'
deadline=time.time()+6*3600
while not (H/'evidence/FINAL-EVIDENCE-READY.json').exists():
 if time.time()>deadline:raise SystemExit('final evidence not ready; no automatic relaunch')
 time.sleep(30)
freeze=json.loads((H/'freeze-g4.json').read_text())
for name,digest in freeze['files'].items():assert hashlib.sha256((J/name).read_bytes()).hexdigest()==digest,name
py=str(J/'venv/bin/python')
subprocess.run([py,str(H/'check-qualification.py'),str(J),'--output',str(J/'mechanical-qualification.json')],check=True)
subprocess.run([py,str(H/'export-job.py'),str(J),str(P/'g4')],check=True)
subprocess.run(['/usr/bin/python3',str(P/'extract-sessions.py'),str(J),'--output',str(P/'g4/session-diagnosis.json')],check=True)
first=subprocess.run([py,str(H/'classify-first-answers.py'),str(J/'experiment')],capture_output=True,text=True,check=True)
(P/'g4/first-answers.json').write_text(json.dumps({'sessions':[json.loads(l) for l in first.stdout.splitlines() if l.strip()]},indent=1)+'\n')
subprocess.run(['/usr/bin/python3',str(H/'build-provider-evidence-amended.py'),str(P)],check=True)
shutil.copy(H/'freeze-g4.json',P/'freeze-record.json')
shutil.copy(H/'evidence/smoke-summary.json',P/'smoke-summary.json')
for name in ('smoke-activation-verification.json','smoke-accounting-diagnosis.json'):
 shutil.copy(H/'evidence'/name,P/name)
for name in ('g4-amendment.json','amendment-post.json','amendment-on-51.md'):
 shutil.copy(H/name,P/name)
for name in ('launch-g4-amended.py','verify-smoke-activation.py','build-provider-evidence-amended.py','export-final-data-amended.py'):
 shutil.copy(H/name,P/(name+'.txt'))
subprocess.run(['/usr/bin/python3',str(P/'reproduce.py'),'--write','--output',str(H/'g4-summary.json')],check=True)
(H/'evidence/PUBLIC-DATA-READY.json').write_text(json.dumps({'ready':True})+'\n')
