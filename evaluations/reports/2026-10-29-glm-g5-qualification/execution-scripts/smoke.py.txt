 checks.append({'run_id':row['run_id'],'status':row['status'],'accounting_complete':row['accounting_complete'],'attempts':len(row['attempts']),'replay_exit':replay_run(root).exit_code,'provenance_sha256':hashlib.sha256((root/'artifacts.json').read_bytes()).hexdigest()})
s={'scrutare_version':__version__,'started_utc':started,'ended_utc':utc(),'seconds':round(time.monotonic()-t,1),'status':r['status'],'smoke_jobs':1,'logical_cells':3,'why_three_cells':'The released evaluator always executes three repeats; one smoke job exercises its actual rule A mode, rather than labelling a direct engine call as recovery-enabled. print(json.dumps({'smoke_complete':True,'cells':3,'cumulative_usage':r['cumulative_usage']}))
