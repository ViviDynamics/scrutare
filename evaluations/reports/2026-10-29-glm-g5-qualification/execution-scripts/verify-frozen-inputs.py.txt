"""Rehash the frozen corpus/wheel/config/scripts and pinned nare, with no model access."""
import hashlib,json,subprocess,zipfile
from pathlib import Path
H=Path(__file__).resolve().parent;J=H/'g5';sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
f=json.loads((H/'freeze-g5.json').read_text());p=json.loads((J/'preparation.json').read_text())
for name,digest in f['files'].items():assert sha(J/name)==digest,name
for name,digest in f['script_sha256'].items():assert sha(H/name)==digest,name
for name,digest in p['input_sha256'].items():assert sha(J/name)==digest,name
wheel,=J.glob('scrutare-*.whl')
with zipfile.ZipFile(wheel)as z:expected={name:hashlib.sha256(z.read(name)).hexdigest()for name in z.namelist()if name.startswith('scrutare/')and not name.endswith('/')}
code="import hashlib,importlib.metadata as m,json;d=m.distribution('scrutare');print(json.dumps({'version':d.version,'files':{str(f):hashlib.sha256(d.locate_file(f).read_bytes()).hexdigest()for f in d.files if str(f).startswith('scrutare/')and not str(f).endswith('.pyc')}}))"
installed=json.loads(subprocess.check_output([str(J/'venv/bin/python'),'-I','-c',code],text=True))
assert installed['version']==p['release_tag'] and installed['files']==expected
n=p['nare_runtime'];checkout=Path(n['console']).parents[2]
assert subprocess.check_output(['git','-C',str(checkout),'rev-parse','HEAD'],text=True).strip()==n['commit']=='96672e1e6d23920fc6cdeebfcad7250c8c36cc5e'
assert subprocess.check_output([n['console'],'--version'],text=True).strip()==n['version']=='2026.10.12'
assert sha(checkout/'uv.lock')==n['uv_lock_sha256']
for name,digest in n['nare_package_sha256'].items():assert sha(checkout/'src'/name)==digest,name
assert set(n['nare_package_sha256'])=={x.relative_to(checkout/'src').as_posix()for x in(checkout/'src/nare').rglob('*.py')}
print(json.dumps({'frozen_inputs_and_scripts_verified':True,'nare_version':'2026.10.12','qualification_rule_a':p['qualification_rule_a'],'model_alias':p['model_alias']}))
