"""Prepare g5 from released rule A support and the frozen g4 profile."""
import argparse,hashlib,json,subprocess,tempfile
from pathlib import Path
H=Path(__file__).resolve().parent;ROOT=Path('/home/jason/Workspace/ViviDynamics/scrutare')
p=argparse.ArgumentParser();p.add_argument('--release-tag',required=True);args=p.parse_args()
assert not (H/'g5').exists(), 'never prepare or launch a second g5'
subprocess.run(['git','fetch','origin','main','--tags'],cwd=ROOT,check=True,capture_output=True)
source=subprocess.check_output(['git','rev-parse',args.release_tag+'^{commit}'],cwd=ROOT,text=True).strip()
with tempfile.TemporaryDirectory() as directory:
 subprocess.run(['gh','release','download',args.release_tag,'--repo','ViviDynamics/scrutare','--pattern','scrutare-*.whl','--dir',directory],check=True,capture_output=True)
 wheel,=Path(directory).glob('scrutare-*.whl');digest=hashlib.sha256(wheel.read_bytes()).hexdigest()
subprocess.run(['/usr/bin/python3',str(H/'make-gate-g5.py'),'--source',source,'--release-tag',args.release_tag],check=True)
old=json.loads((H.parent/'glm-g4-2026-10-28/g4/preparation.json').read_text())
command=['/usr/bin/python3',str(H/'prepare-job.py'),'g5','--source',source,'--release-tag',args.release_tag,'--wheel-sha256',digest,'--corpus',old['corpus'].removeprefix('evaluations/'),'--cases',*old['cases'],'--comparison-set',old['comparison_set'],'--ceiling',str(old['configured_max_tokens']),'--model-alias',old['model_alias'],'--timeout-seconds',str(old['timeout_seconds']),'--concurrency',str(old['concurrency']),'--transient-retries','0','--retry-delay',str(old['transient_retry_delay_seconds']),'--retry-budget','0','--gate',str(H/'gate-g5.json'),'--related-paths',*old['related_paths'],'--assessment-tokens',str(old['assessment_tokens']),'--structured-output',old['structured_output'],'--max-output-tokens',str(old['max_output_tokens']),'--line-counts','--citation-correction','--protocol-correction','--qualification-rule-a']
assert old['variants'] is None
subprocess.run(command,check=True)
print('Prepared without freezing; run freeze.py after final independent review.')
