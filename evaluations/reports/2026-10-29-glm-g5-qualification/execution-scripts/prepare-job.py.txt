"""Freeze one evaluation job before any provider call. No credential is read here."""
import argparse
import hashlib
import json
import shutil
import subprocess
import tempfile
from pathlib import Path

ROOT = Path('/home/user/Workspace/ViviDynamics/scrutare')
STATE = Path(__file__).resolve().parent
NARE = Path('/home/user/Workspace/ViviDynamics/scrutare/.agents/state/nare-2026.10.12')
# LiteLLM's configured attribution rate for the self-hosted group (model_info output_cost_per_token).
# No provider bills it; it is recorded only so the proxy's own cost fields can be explained.
RATE_OUTPUT_PER_TOKEN = 5e-07

parser = argparse.ArgumentParser()
parser.add_argument('label')
parser.add_argument('--source', required=True, help='full commit SHA reachable from origin/main')
parser.add_argument('--corpus', required=True, help='corpus directory name under evaluations/')
parser.add_argument('--cases', required=True, nargs='+')
parser.add_argument('--comparison-set', required=True)
parser.add_argument('--variants', default=None)
parser.add_argument('--ceiling', type=int, default=100000)
parser.add_argument('--model-alias', required=True)
parser.add_argument('--timeout-seconds', type=int, required=True)
parser.add_argument('--concurrency', type=int, required=True)
parser.add_argument('--transient-retries', type=int, required=True)
parser.add_argument('--retry-delay', type=float, required=True)
parser.add_argument('--retry-budget', type=int, required=True)
parser.add_argument('--gate', required=True, help='JSON file with the frozen gate declaration')
parser.add_argument('--related-paths', nargs='*', default=None)
parser.add_argument('--ruff', default=None,
                    help='absolute pinned Ruff 0.16.10; captures real evidence for every case')
parser.add_argument('--assessment-tokens', type=int, default=None,
                    help='explicit assessor reservation (needs a source that has the option)')
parser.add_argument('--structured-output', choices=['json_schema'], default=None,
                    help='opt the default rail in to schema-constrained output (#78)')
parser.add_argument('--max-output-tokens', type=int, default=None,
                    help='per-request output token cap on the default rail (#82)')
parser.add_argument('--release-tag', required=True, help='Scrutare release tag at --source')
parser.add_argument('--citation-correction', action='store_true')
parser.add_argument('--protocol-correction', action='store_true')
parser.add_argument('--qualification-rule-a', action='store_true')
parser.add_argument('--wheel-sha256', required=True)
parser.add_argument('--line-counts', action='store_true',
                    help='opt in to context.line_counts (needs a source that has it)')
args = parser.parse_args()

job = STATE / args.label
job.mkdir()
sha = subprocess.check_output(['git', 'rev-parse', args.source + '^{commit}'], cwd=ROOT,
                              text=True).strip()
assert sha == args.source
tagged = subprocess.check_output(['git', 'rev-parse', args.release_tag + '^{commit}'], cwd=ROOT,
                                 text=True).strip()
assert tagged == sha, 'the frozen source must be the released commit'
subprocess.run(['git', 'merge-base', '--is-ancestor', sha, 'origin/main'], cwd=ROOT, check=True)
with tempfile.TemporaryDirectory() as temporary:
    tree = Path(temporary) / 'src'
    tree.mkdir()
    archive = subprocess.run(['git', 'archive', sha, 'evaluations/' + args.corpus], cwd=ROOT, check=True, capture_output=True)
    subprocess.run(['tar', '-x', '-C', str(tree)], input=archive.stdout, check=True)
    subprocess.run(['gh', 'release', 'download', args.release_tag, '--repo', 'ViviDynamics/scrutare',
                    '--pattern', 'scrutare-*.whl', '--dir', str(job)], check=True, capture_output=True)
    source_corpus = tree / 'evaluations' / args.corpus
    manifest = json.loads((source_corpus / 'manifest.json').read_text())
    entries = {entry['id']: entry for entry in manifest['cases']}
    selected = [entries[name] for name in args.cases]
    assert all(entry['split'] == 'development' for entry in selected)
    corpus = job / 'corpus'
    for entry in selected:
        shutil.copytree(source_corpus / entry['capture'], corpus / entry['capture'])
        (corpus / entry['labels']).parent.mkdir(parents=True, exist_ok=True)
        shutil.copy(source_corpus / entry['labels'], corpus / entry['labels'])
        for side in entry['context_sources'].values():
            shutil.copytree(source_corpus / side['directory'], corpus / side['directory'])
    (corpus / 'manifest.json').write_text(json.dumps(
        {'schema_version': 1, 'cases': selected}, indent=2, sort_keys=True) + '\n')
    related = args.related_paths
    if related is None and (source_corpus / 'related-paths.json').exists():
        related = json.loads((source_corpus / 'related-paths.json').read_text())
    if related is None:
        related = ['client.py', 'policy.py', 'schema.py', 'formatting.py', 'test_formatting.py']
    assert isinstance(related, list) and related and all(
        isinstance(item, str) and item and not any(c.isspace() for c in item)
        for item in related), 'related paths must be separate whitespace-free strings'
wheel, = job.glob('scrutare-*.whl')
assert hashlib.sha256(wheel.read_bytes()).hexdigest() == args.wheel_sha256
subprocess.run(['uv', 'venv', '--python', '3.10', str(job / 'venv')], check=True,
               capture_output=True)
subprocess.run(['uv', 'pip', 'install', '--python', str(job / 'venv/bin/python'), str(wheel)],
               check=True, capture_output=True)
tool_capture = None
if args.ruff:
    # Real pinned-Ruff evidence over the private corpus copy, by the measured wheel's producer.
    code = ('import json,sys\nfrom pathlib import Path\n'
            'from scrutare.evaluation.tool_capture import capture_development_ruff\n'
            'index = capture_development_ruff(Path(sys.argv[1]), Path(sys.argv[2]), '
            'executable=Path(sys.argv[3]))\n'
            'print(json.dumps({k: v["status"] for k, v in index["cases"].items()}))\n')
    out = subprocess.run([str(job / 'venv/bin/python'), '-c', code, str(corpus),
                          str(job / 'tool-captures'), args.ruff], check=True,
                         capture_output=True, text=True).stdout
    index = json.loads((job / 'tool-captures/index.json').read_text())
    manifest = json.loads((corpus / 'manifest.json').read_text())
    (corpus / 'analysis').mkdir()
    for entry in manifest['cases']:
        record = index['cases'][entry['id']]
        assert record['status'] == 'complete', (entry['id'], record['status'])
        shutil.copy(job / 'tool-captures' / record['capture'],
                    corpus / 'analysis' / record['capture'])
        entry['analysis_capture'] = 'analysis/' + record['capture']
    (corpus / 'manifest.json').write_text(json.dumps(manifest, indent=2, sort_keys=True) + '\n')
    tool_capture = {'tool': index['tool'], 'statuses': json.loads(out),
                    'diagnostics_per_case': {
                        e['id']: len(json.loads((corpus / e['analysis_capture']).read_text())[
                            'diagnostics']) for e in manifest['cases']},
                    'index_sha256': hashlib.sha256(
                        (job / 'tool-captures/index.json').read_bytes()).hexdigest()}
context = {'enabled': True, 'related_paths': related}
if args.line_counts:
    context['line_counts'] = True
(job / 'evaluation.json').write_text(json.dumps({
    'models': {'default': {'provider': 'openai',
                           'base_url': 'https://litellm.vividynamics.com/v1',
                           'model': args.model_alias,
                           **({'structured_output': args.structured_output}
                              if args.structured_output else {}),
                           **({'max_output_tokens': args.max_output_tokens}
                              if args.max_output_tokens else {})}},
    'budgets': {'per_persona_tokens': args.ceiling, 'review_max_tokens': args.ceiling},
    'rounds': {'max': 3},
    'context': context,
    **({'findings': {'evidence': 'v2', 'citation_correction': True, **({'protocol_correction': True} if args.protocol_correction else {})}} if args.citation_correction else {})}, indent=2) + '\n')
shutil.copy(args.gate, job / 'gate.json')
variants = args.variants.split(',') if args.variants else None
# The runtime is part of the freeze: version, release commit, contract and console bytes.
nare_console = NARE / '.venv/bin/nare'
nare_runtime = {
    'console': str(nare_console),
    'version': subprocess.check_output([str(nare_console), '--version'], text=True).strip(),
    'commit': subprocess.check_output(['git', '-C', str(NARE), 'rev-parse', 'HEAD'], text=True).strip(),
    'contract': json.loads(subprocess.check_output([str(nare_console), 'contract'], text=True)),
    'uv_lock_sha256': hashlib.sha256((NARE / 'uv.lock').read_bytes()).hexdigest(),
    'nare_package_sha256': {p.relative_to(NARE / 'src').as_posix(): hashlib.sha256(p.read_bytes()).hexdigest()
                            for p in sorted((NARE / 'src/nare').rglob('*.py'))},
}
assert nare_runtime['version'] == '2026.10.12'
assert nare_runtime['commit'] == '96672e1e6d23920fc6cdeebfcad7250c8c36cc5e'
if args.structured_output:
    assert args.structured_output in nare_runtime['contract']['structured_output']['modes']
gate = json.loads((job / 'gate.json').read_text())
cells = gate['expected_runs']
inputs = {path.relative_to(job).as_posix(): hashlib.sha256(path.read_bytes()).hexdigest()
          for path in sorted([*corpus.rglob('*'), job / 'evaluation.json', job / 'gate.json',
                              *(job / 'tool-captures').glob('*'), wheel]) if path.is_file()}
(job / 'preparation.json').write_text(json.dumps({
    'label': args.label, 'source_commit': sha, 'source_is_ancestor_of_origin_main': True,
    'release_tag': args.release_tag, 'structured_output': args.structured_output,
    'max_output_tokens': args.max_output_tokens,
    'nare_runtime': nare_runtime,
    'model_alias': args.model_alias, 'backend_revision_verified': False,
    'corpus': f'evaluations/{args.corpus}', 'cases': args.cases, 'split': 'development',
    'comparison_set': args.comparison_set, 'variants': variants, 'repeats': 3,
    'expected_runs': cells, 'configured_max_tokens': args.ceiling, 'concurrency': args.concurrency,
    'max_turns': 6, 'timeout_seconds': args.timeout_seconds,
    'transient_retries_per_cell': args.transient_retries,
    'transient_retry_delay_seconds': args.retry_delay,
    'transient_retry_budget_job': args.retry_budget,
    'nominal_estimate_usd': {
        'method': 'attempts x configured ceiling x LiteLLM attribution output rate for the self-hosted group; external spend is USD 0 by construction (no paid deployment or fallback is reachable)',
        'output_rate_per_token': RATE_OUTPUT_PER_TOKEN,
        'first_attempts': round(cells * args.ceiling * RATE_OUTPUT_PER_TOKEN, 2),
        'worst_case_with_all_budgeted_reruns': round(
            (cells * 2 if args.qualification_rule_a else cells + args.retry_budget) * args.ceiling * RATE_OUTPUT_PER_TOKEN, 2),
        'not_a_billing_cap': True},
    'assessment_tokens': args.assessment_tokens,
    'tool_capture': tool_capture, 'context_line_counts': args.line_counts,
    'citation_correction': args.citation_correction, 'qualification_rule_a': args.qualification_rule_a, 'protocol_correction': args.protocol_correction, 'wheel_origin': 'GitHub released asset',
    'related_paths': related,
    'holdout_used': False, 'frozen_before_this_job_calls': True,
    'input_sha256': inputs}, indent=2) + '\n')
print(json.dumps({'job': str(job), 'cells': cells, 'inputs': len(inputs)}))
