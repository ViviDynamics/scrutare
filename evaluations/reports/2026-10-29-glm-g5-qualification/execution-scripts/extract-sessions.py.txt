"""Export session metadata from private g4 captures, including correction and retry streams.

No credentials or transcript content are exported. Run against the retained job after
it exits; this reader does not modify any measured artifact.
"""
import argparse
import hashlib
import json
import re
from pathlib import Path


def session_identity(relative):
    parts = relative.parts
    offset = 5 if parts[0] == '_retries' else 3
    if len(parts) != offset + 4 or parts[offset] != 'sessions' or parts[-1] != 'result.json':
        raise ValueError('unexpected session path')
    if offset == 5:
        if not re.fullmatch(r'attempt-\d{4}', parts[4]):
            raise ValueError('unexpected cell attempt path')
        cell, number = '/'.join(parts[1:4]), int(parts[4].split('-')[1])
    else:
        cell, number = '/'.join(parts[:3]), 1
    return cell, number, '/'.join(parts[:offset]), parts[offset + 1], parts[offset + 2]


def events(path):
    if not path.exists():
        return []
    result = []
    for line in path.read_bytes().splitlines():
        try:
            result.append(json.loads(line))
        except ValueError:
            continue
    return result


def request_outputs(directory):
    def costs(name):
        return [e['detail']['output'] for e in events(directory / name) if e.get('type') == 'cost']
    initial, correction = costs('stdout.jsonl'), costs('correction.stdout.jsonl')
    return {'initial': initial, 'correction': correction, 'all': initial + correction}


def extract(job):
    experiment = job / 'experiment'
    rows, digests = [], []
    for result in sorted(experiment.rglob('sessions/*/attempt-*/result.json')):
        relative = result.relative_to(experiment)
        cell, cell_attempt, prefix, persona, attempt = session_identity(relative)
        data = json.loads(result.read_text())
        first = events(result.parent / 'stdout.jsonl')
        correction = events(result.parent / 'correction.stdout.jsonl')
        terminal = next((e for e in reversed(first) if e.get('type') == 'result'), {})
        corrected = next((e for e in reversed(correction) if e.get('type') == 'result'), None)
        costs = request_outputs(result.parent)
        def constrained(stream):
            return sum(e.get('type') == 'progress' and
                       'structured_output' in (e.get('detail') or {})
                       for e in stream)
        structured = data.get('structured_output') or {}
        if structured.get('constrained_turns') not in (None, constrained(first)):
            raise ValueError('constrained count disagrees with first stream')
        def optional(name):
            path = result.parent / name
            return json.loads(path.read_text()) if path.exists() else None
        request = optional('protocol-correction.json')
        validation = optional('citation-validation.json')
        before = terminal.get('usage')
        error = terminal.get('error')
        answer = terminal.get('output')
        findings = answer.get('findings') if isinstance(answer, dict) else None
        assessments = answer.get('assessments') if isinstance(answer, dict) else None
        anchors = None if not isinstance(findings, list) else [
            {'file': f.get('file') if isinstance(f.get('file'), str) and
             re.fullmatch(r'[A-Za-z0-9_./-]+', f['file']) else '<redacted-non-path>',
             'line': f.get('line') if type(f.get('line')) is int else None,
             'side': f.get('side') if f.get('side') in ('RIGHT', 'LEFT') else None}
            for f in findings if isinstance(f, dict)]
        identifiers = None if not isinstance(assessments, list) else [
            a['candidate_id'] if isinstance(a.get('candidate_id'), str) and
            re.fullmatch(r'[0-9a-f]{64}', a['candidate_id']) else 'invalid-candidate-id'
            for a in assessments if isinstance(a, dict)]
        corrected_output = corrected.get('output') if isinstance(corrected, dict) else None
        corrected_findings = (corrected_output.get('findings')
                              if isinstance(corrected_output, dict) else None)
        corrected_assessments = (corrected_output.get('assessments')
                                 if isinstance(corrected_output, dict) else None)
        correction_anchors = None if not isinstance(corrected_findings, list) else [
            {'line': f.get('line') if type(f.get('line')) is int else None}
            for f in corrected_findings if isinstance(f, dict)]
        correction_ids = None if not isinstance(corrected_assessments, list) else [
            a['candidate_id'] if isinstance(a.get('candidate_id'), str) and
            re.fullmatch(r'[0-9a-f]{64}', a['candidate_id']) else 'invalid-candidate-id'
            for a in corrected_assessments if isinstance(a, dict)]
        rows.append({
            'correction_answer_anchors': correction_anchors,
            'correction_assessment_candidate_ids': correction_ids,
            'cell': cell, 'cell_attempt': cell_attempt, 'cell_attempt_prefix': prefix,
            'persona': persona, 'attempt': attempt, 'status': data['status'],
            'finding_answer_anchors': anchors, 'assessment_answer_candidate_ids': identifiers,
            'reason': data['reason'], 'tokens': data['usage']['total'],
            'input_tokens': data['usage']['input'], 'output_tokens': data['usage']['output'],
            'nare_status': terminal.get('status'), 'nare_stop_reason': terminal.get('stop_reason'),
            'nare_turns': terminal.get('turns'),
            'first_run_tokens': sum(before[k] for k in
                                    ('input', 'output', 'cache_read', 'cache_write'))
            if before else None,
            'citation_validation_reason': validation['reason'] if validation else None,
            'nare_error': re.sub(r'used \d+, limit \d+', 'used N, limit N', error)
            if error else None,
            'structured_output_mode': structured.get('mode'),
            'constrained_turns': structured.get('constrained_turns'),
            'correction_constrained_turns': constrained(correction),
            'max_output_tokens': data.get('max_output_tokens'),
            'protocol_correction': data.get('protocol_correction'),
            'protocol_correction_request': {k: request[k] for k in ('refusal', 'defects')}
            if request else None,
            'correction_terminal': {k: corrected.get(k)
                                    for k in ('status', 'stop_reason', 'turns', 'usage')}
            if corrected else None,
            'initial_request_output_tokens': costs['initial'],
            'correction_request_output_tokens': costs['correction'],
            'request_output_tokens': costs['all']})
        for path in sorted(result.parent.iterdir()):
            if path.is_file():
                digests.append([path.relative_to(experiment).as_posix(),
                                hashlib.sha256(path.read_bytes()).hexdigest()])
    return {'schema_version': 2, 'job': job.name, 'sessions': rows,
            'private_records_sha256': hashlib.sha256(json.dumps(digests).encode()).hexdigest(),
            'private_records': len(digests),
            'source': 'retained result and initial/correction streams and refusal records'}


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('job', type=Path)
    parser.add_argument('--output', type=Path, required=True)
    args = parser.parse_args()
    document = extract(args.job)
    text = json.dumps(document, indent=1, sort_keys=True) + '\n'
    if '/home/' in text or '/tmp/' in text:
        raise ValueError('local path in public session metadata')
    args.output.write_text(text)
    print(json.dumps({'sessions': len(document['sessions'])}))


if __name__ == '__main__':
    main()
