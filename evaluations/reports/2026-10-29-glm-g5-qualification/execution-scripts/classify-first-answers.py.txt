"""Reconcile every first answer and its protocol-defect inventory. No model calls or text export."""
import json,sys
from pathlib import Path
from scrutare.engine.session_output import _decode_evidence
from scrutare.findings.protocol_correction import protocol_context,protocol_defects
from scrutare.findings.evidence import CitationValidationError
exp=Path(sys.argv[1])
for path in sorted(exp.rglob('sessions/*/attempt-*/result.json')):
 attempt=path.parent;data=json.loads(path.read_text())
 if 'protocol_correction' not in data:continue
 inv=json.loads((attempt/'invocation.json').read_text());schema=json.loads((attempt/'findings.schema.json').read_text())
 context=protocol_context(schema,inv['argv'][2]);root=attempt.parent.parent.parent/'review-inputs'
 row={'session':attempt.relative_to(exp).as_posix(),'kind':context.kind,'status':data['status'],'reason':data['reason'],'first_answer':None,'correctable':None,'first_reconciled':False}
 try:
  first=_decode_evidence((attempt/'stdout.jsonl').read_bytes(),(attempt/'session.json').read_bytes(),exit_code=data['exit_code'],expected_limit=data['invocation_limit'],expected_root=root)
  row['first_reconciled']=True
  if (first.status=='done' and not first.cut_off
      and first.nare_version==inv['nare_version'] and first.contract==inv['contract']
      and not (first.stop_reason=='budget' and not first.partial)):
   try:
    context.parse(first.output,root,data['persona'],data['artifact_directory']);context.parse(first.saved_output,root,data['persona'],data['artifact_directory']);row['first_answer']='valid'
   except CitationValidationError as error:row['first_answer']='citation:'+error.reason
   except (ValueError,TypeError,KeyError):row['first_answer']='protocol'
   if row['first_answer']!='valid' and first.usage.total<data['invocation_limit'] and json.dumps(first.output,sort_keys=True)==json.dumps(first.saved_output,sort_keys=True):
    found=protocol_defects(first.output,root,context)
    row['correctable']=None if found is None else [d.to_dict() for d in found]
 except (OSError,ValueError,TypeError,KeyError):pass
 print(json.dumps(row,sort_keys=True))
