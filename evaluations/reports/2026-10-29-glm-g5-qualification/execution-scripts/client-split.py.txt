"""Spend-log rows in a UTC window split by client user agent; never prints the key."""
import collections, json, os, sys, urllib.parse, urllib.request
start, end, out = sys.argv[1], sys.argv[2], sys.argv[3]
K = os.environ['K']; rows, page = [], 1
while True:
    q = urllib.parse.urlencode({'start_date': start, 'end_date': end, 'page': page, 'page_size': 100})
    d = json.load(urllib.request.urlopen(urllib.request.Request(
        'https://litellm.vividynamics.com/spend/logs/v2?' + q, headers={'Authorization': 'Bearer ' + K}), timeout=120))
    rows += d['data']
    if page >= d['total_pages']: break
    page += 1
by = {}
for r in rows:
    md = r.get('metadata') or {}
    agent = (md.get('user_agent') or 'none').split('/')[0]
    b = by.setdefault(f"{agent} | {r['model_group']} | {r['api_base']}", {'rows': 0, 'prompt_tokens': 0, 'completion_tokens': 0, 'spend': 0.0, 'first': None, 'last': None, 'fallbacks': collections.Counter()})
    b['rows'] += 1; b['prompt_tokens'] += r['prompt_tokens'] or 0; b['completion_tokens'] += r['completion_tokens'] or 0
    b['spend'] = round(b['spend'] + (r['spend'] or 0), 6)
    t = r.get('startTime'); b['first'] = min(filter(None, [b['first'], t])); b['last'] = max(filter(None, [b['last'], t]))
    b['fallbacks'][str(md.get('attempted_fallbacks'))] += 1
for b in by.values(): b['fallbacks'] = dict(b['fallbacks'])
doc = {'window_utc': [start, end], 'rows': len(rows), 'by_client_group_base': by}
open(out, 'w').write(json.dumps(doc, indent=1) + '\n'); print(json.dumps(doc, indent=1))
