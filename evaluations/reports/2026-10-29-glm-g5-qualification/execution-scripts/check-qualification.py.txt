 r=inspect_rule_a(a.job/'experiment',json.loads((a.job/'gate.json').read_text()),ledger)
except (OSError,ValueError,KeyError,TypeError,IndexError) as error:
 try:experiment=json.loads((a.job/'experiment/experiment.json').read_text())

 r={'schema_version':1,'rule':'A','tests':{'inspectable_complete_artifacts':False},'qualification':'not_qualified','inspection_error_type':type(error).__name__,'cumulative_usage':experiment.get('cumulative_usage'),'reconciled_cumulative_usage':{k:sum(q['usage'][k] for q in ledger['requests']) for k in keys},'cells':[],'semantic_qualification':False,'holdout_used':False,'limits':'Missing/incomplete evidence fails qualification.  r['tests']['router_'+label]=ok
