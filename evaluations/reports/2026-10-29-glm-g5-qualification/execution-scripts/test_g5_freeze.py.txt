"""Prospective g5 gate/freezer invariants, with temporary offline dummy releases."""
import hashlib
import json
import shutil
import subprocess
import sys
from pathlib import Path

import pytest

H = Path(__file__).resolve().parent
OLD = H.parent / 'glm-g4-2026-10-28/g4'


def prepare(tmp_path):
    state = tmp_path / 'state'
    old = state / 'glm-g4-2026-10-28/g4'
    old.mkdir(parents=True)
    new = state / H.name
    new.mkdir()
    for name in ('gate.json', 'preparation.json', 'evaluation.json'):
        shutil.copy2(OLD / name, old / name)
    for name in ('make-gate-g5.py', 'freeze.py', 'rule-a-common-predicates.md'):
        shutil.copy2(H / name, new / name)
    subprocess.run([sys.executable, str(new / 'make-gate-g5.py'), '--source', '0' * 40,
                    '--release-tag', '2026.10.999'], check=True, capture_output=True)
    job = new / 'g5'
    job.mkdir()
    shutil.copy2(new / 'gate-g5.json', job / 'gate.json')
    shutil.copy2(old / 'evaluation.json', job / 'evaluation.json')
    shutil.copytree(OLD / 'corpus', job / 'corpus')
    wheel = job / 'scrutare-2026.10.999-py3-none-any.whl'
    wheel.write_bytes(b'offline dummy, never executed or installed')
    prep = json.loads((old / 'preparation.json').read_text())
    prep.update(qualification_rule_a=True, label='g5', source_commit='0' * 40,
                release_tag='2026.10.999', transient_retries_per_cell=0,
                transient_retry_budget_job=0)
    prep['input_sha256'] = {p.relative_to(job).as_posix(): hashlib.sha256(p.read_bytes()).hexdigest()
                            for p in [*job.rglob('*')] if p.is_file()}
    (job / 'preparation.json').write_text(json.dumps(prep))
    return new, job


def test_prospective_gate_changes_only_the_authorized_policy(tmp_path):
    new, job = prepare(tmp_path)
    gate = json.loads((job / 'gate.json').read_text())
    old = json.loads((OLD / 'gate.json').read_text())
    for key in gate['g4_unchanged_profile_keys']:
        assert gate[key] == old[key]
    assert gate['qualification_rule_a'] is True
    assert gate['whole_cell_rerun']['max_reruns_per_cell'] == 1
    assert gate['whole_cell_rerun']['max_cell_attempts'] == 72
    assert gate['all_defect_case_repeats_nonempty_valid_v2'] == 24
    assert gate['transient_retry']['per_cell'] == gate['transient_retry']['job_budget'] == 0
    assert gate['smoke_timeout_seconds'] == gate['timeout_seconds'] == 1800
    assert gate['smoke_completion_required'] is True
    assert 'without an idle window' in gate['launch_condition']
    assert '2026-10-25' in gate['launch_condition'] and '2026-10-28' in gate['launch_condition']
    assert gate['schedules_allowed'].startswith('g5 only.')
    result = subprocess.run([sys.executable, str(new / 'freeze.py')], capture_output=True)
    assert result.returncode == 0, result.stderr
    assert json.loads((new / 'freeze-g5.json').read_text())['g4_corpus_hashes_identical'] is True


@pytest.mark.parametrize('mutation', ['budget', 'corpus', 'evaluation', 'already_launched'])
def test_freezer_refuses_profile_drift_or_reuse(tmp_path, mutation):
    new, job = prepare(tmp_path)
    if mutation == 'budget':
        gate = json.loads((job / 'gate.json').read_text())
        gate['max_tokens_per_run'] += 1
        (job / 'gate.json').write_text(json.dumps(gate))
    elif mutation == 'corpus':
        p = next((job / 'corpus').rglob('diff.patch'))
        p.write_text(p.read_text() + '\n')
    elif mutation == 'evaluation':
        config = json.loads((job / 'evaluation.json').read_text())
        config['rounds']['max'] += 1
        (job / 'evaluation.json').write_text(json.dumps(config))
    else:
        (job / 'process.json').write_text('{}')
    result = subprocess.run([sys.executable, str(new / 'freeze.py')], capture_output=True)
    assert result.returncode != 0
    assert not (new / 'freeze-g5.json').exists()
